{{/* 右侧栏卡片组。 顺序固定为：分类 → 标签 → 最新文章 → 友情链接（→ 其它外链）。 原站是 JS 硬编码，这里全部由 Hugo 数据生成。 */}}
{{- /* 1. 分类 */ -}} {{- with site.Taxonomies.categories }}
  • 分类
    • {{- range .ByCount }} {{- $c := .Page.Params.style.background -}}
  • {{- with $c }}{{ end }} {{ .Page.Title }} {{ .Count }}
    • {{- end }}
{{- end }} {{- /* 2. 标签 */ -}}
  • 标签
    • {{- $tags := site.Taxonomies.tags -}} {{- if $tags }} {{- range $tags.ByCount }}
  • {{ .Page.Title }} {{ .Count }}
    • {{- end }} {{- else }}
  • 还没有标签
    • {{- end }}
{{- /* 3. 最新文章：若存在 pinned / recommend 标记则改称「推荐文章」 */ -}} {{- $pinned := where site.RegularPages "Params.pinned" true -}} {{- $recommend := where site.RegularPages "Params.recommend" true -}} {{- $picks := $pinned | append $recommend | uniq -}} {{- $isPicks := gt (len $picks) 0 -}} {{- if not $isPicks }} {{- $picks = first 5 (where site.RegularPages "Section" "post") -}} {{- end -}} {{- if $picks }}
  • {{- if $isPicks }} 推荐文章 {{- else }} 最新文章 {{- end }}
    • {{- range first 5 $picks }}
  • {{ .Title }}
    • {{- end }}
{{- end }} {{- /* 4. 友情链接 */ -}} {{- with site.Params.links }}
  • 友情链接
    • {{- range . }}
  • {{ .name }}
    • {{- end }}
{{- end }} {{- /* 5. 其它外链：原本挤在顶栏「其他」下拉里，去掉下拉后移到侧栏 */ -}} {{- with site.Params.socials }}
  • 其它
    • {{- range . }}
  • {{ .name }}
    • {{- end }}
{{- end }}
