{{- /* 3. 推荐文章：用标签驱动。 在 params.toml 的 [recommend] tag 里指定标签名（默认「推荐文章」）， 给文章 front matter 的 tags 加上这个标签，它就会出现在这里。 一篇都没打该标签时，回落显示最新文章，卡片不会空着。 */ -}} {{- $recTag := site.Params.recommend.tag | default "推荐文章" -}} {{- $recommended := where site.RegularPages "Params.tags" "intersect" (slice $recTag) -}} {{- $recommended = $recommended.ByDate.Reverse -}} {{- $isRec := gt (len $recommended) 0 -}} {{- $picks := $recommended -}} {{- if not $isRec }} {{- $picks = first 5 (where site.RegularPages "Section" "post") -}} {{- end -}} {{- if $picks }}
  • {{- /* 标题不指向 /tags/推荐文章/：这个标签是内部驱动用的， 不应作为一个可发现的标签页入口出现在全站每一页。 卡片的实际价值是下面那几条文章链接。 */ -}} {{- if $isRec }} 推荐文章 {{- else }} 最新文章 {{- end }}
    • {{- range first 5 $picks }}
  • {{ .Title }}
    • {{- end }}
{{- end }} {{- /* 4. 友情链接 */ -}} {{- with site.Params.links }}
