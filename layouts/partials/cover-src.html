{{- /* 解析封面图，并在构建期做尺寸缩放与格式转换。 返回 dict：src（1x 地址）、srcset（1x/2x，可能为空）、alt。 找不到图时返回空 src 的 dict，调用方判空即可。 为什么要处理： 站内封面原本是 1920x1080 的 PNG，单张最大 4.5MB，9 张合计 17MB， 而卡片上只显示 228x152 像素——等于为了几个小图下载十几 MB。 处理策略： - 页面资源（page bundle 内的图）交给 Hugo 处理 - 外链 URL 原样保留，不做处理（拿不到文件） - 处理失败时回退原图，不让构建因为一张图挂掉 - 宽度取 2x 与 1x 两档生成 srcset，兼顾高分屏 参数：page（必填）、width（1x 目标宽，默认 480）、 width2x（2x 目标宽，默认 960）、quality（默认 80） */ -}} {{- $page := .page -}} {{- $w1 := .width | default 480 -}} {{- $w2 := .width2x | default 960 -}} {{- $q := .quality | default 80 -}} {{- $name := $page.Params.image -}} {{- $src := "" -}} {{- $srcset := "" -}} {{- if $name -}} {{- if or (hasPrefix $name "http://") (hasPrefix $name "https://") -}} {{- $src = $name -}} {{- else -}} {{- with $page.Resources.GetMatch $name -}} {{- $img := . -}} {{- if in (slice "image" "images") .ResourceType -}} {{- $o1 := $img -}} {{- $o2 := $img -}} {{- /* 一律转 WebP，宽度只缩不放。 目标比原图宽时（例如详情页要 960，而封面只有 640 宽）， 仍按原宽转 WebP——否则会退回原始 PNG， 实测 640x360 的 cover-05.png 就是这样漏过处理的。 */ -}} {{- if ge $img.Width $w1 -}} {{- $o1 = $img.Resize (printf "%dx webp q%d" $w1 $q) -}} {{- else -}} {{- $o1 = $img.Resize (printf "%dx webp q%d" $img.Width $q) -}} {{- end -}} {{- if ge $img.Width $w2 -}} {{- $o2 = $img.Resize (printf "%dx webp q%d" $w2 $q) -}} {{- else -}} {{- $o2 = $img.Resize (printf "%dx webp q%d" $img.Width $q) -}} {{- end -}} {{- $src = $o1.RelPermalink -}} {{- if ne $o1.RelPermalink $o2.RelPermalink -}} {{- $srcset = printf "%s 1x, %s 2x" $o1.RelPermalink $o2.RelPermalink -}} {{- end -}} {{- else -}} {{- /* 非图片资源或无法处理，回退原地址 */ -}} {{- $src = $img.RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return (dict "src" $src "srcset" $srcset "alt" ($page.Title | plainify)) -}}
